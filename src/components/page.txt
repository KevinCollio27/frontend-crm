"use client"

import { useState } from "react"

import {
  ChannelsSection,
  GeneralSection,
  ProfessionalSection,
} from "@/components/contact-form/contact-form-sections"
import { CrmBackdrop } from "@/components/contact-form/crm-backdrop"
import { useContactForm } from "@/components/contact-form/use-contact-form"
import { Button } from "@/components/ui/button"

import {
  Sheet,
  SheetClose,
  SheetContent,
  SheetDescription,
  SheetFooter,
  SheetHeader,
  SheetTitle,
  SheetTrigger,
} from "@/components/ui/sheet"

export default function SheetIdeaPage() {
  const form = useContactForm()
  const [open, setOpen] = useState(false)

                <GeneralSection form={form} />
                <FieldSeparator />
                <ChannelsSection form={form} />
                <FieldSeparator />
                <ProfessionalSection form={form} />
              </div>
            </div>

              />
              <Button onClick={() => setOpen(false)}>Guardar contacto</Button>
            </SheetFooter>
          </SheetContent>
        </Sheet>
      }
    />
